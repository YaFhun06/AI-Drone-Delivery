\subsection{Functional Requirements}

\begin{longtable}{@{}p{4cm}p{9cm}@{}}
\toprule
\textbf{Module} & \textbf{Functions} \\
\midrule
\endhead
\textbf{Module 1. User Management} &
\begin{enumerate}
  \item Register an account.
  \item Log in and log out.
  \item Reset forgotten passwords.
  \item Manage user profiles
  \item Manage user roles.
  \item Manage user permissions.
\end{enumerate} \\
\addlinespace
\textbf{Module 2. Customer \& Delivery Management} &
\begin{enumerate}
  \item Manage customer information.
  \item Manage delivery addresses.
  \item Create, view, update, and cancel delivery orders.
  \item Track delivery order status.
  \item Manage package information (weight, dimensions, description, etc.).
\end{enumerate} \\
\addlinespace
\textbf{Module 3. Delivery Workflow Management} &
\begin{enumerate}
  \item Review and approve delivery orders.
  \item Reject delivery orders.
  \item Schedule deliveries.
  \item Assign orders to landing stations.
  \item Monitor delivery progress.
  \item Update delivery status.
  \item Handle failed deliveries.
  \item Complete delivery orders.
\end{enumerate} \\
\addlinespace
\textbf{Module 4. Landing Station Management} &
\begin{enumerate}
  \item Manage landing stations.
  \item Manage landing station locations and operational status.
  \item Monitor landing station capacity.
  \item Confirm package receipt.
  \item Confirm drone pickup and return.
  \item Update package status at the landing station.
\end{enumerate} \\
\addlinespace
\textbf{Module 5. Dashboard \& AI Services} &
\begin{enumerate}
  \item Display an overview dashboard.
  \item View delivery order statistics.
  \item View revenue statistics.
  \item Monitor successful and failed delivery rates.
  \item Monitor landing station performance.
  \item Predict Estimated Time of Arrival (ETA).
  \item Provide AI chatbot support.
  \item Generate AI delivery summaries.
  \item Perform AI-powered analytics and delivery trend analysis.
\end{enumerate} \\
\bottomrule
\end{longtable}
