\section{Mục tiêu dự án}

\begin{itemize}
\item
  \textbf{Nền tảng cốt lõi:} Xây dựng nền tảng quản lý giao hàng bằng drone, bao quát năm module chính gồm đơn hàng, kiện hàng, trạm, theo dõi và phân tích có hỗ trợ AI.
\item
  \textbf{Tự động hóa quy trình:} Tự động hóa toàn bộ vòng đời giao hàng, từ tạo yêu cầu giao hàng đến xác nhận giao hàng thành công.
\item
  \textbf{Tích hợp AI:} Ứng dụng AI vào dự đoán thời gian giao hàng (ETA), tóm tắt thông tin giao hàng và hỗ trợ chatbot.
\item
  \textbf{Hỗ trợ đa nền tảng:} Cung cấp khả năng truy cập thông qua cổng web, ứng dụng di động và RESTful API.
\item
  \textbf{Kiến trúc kỹ thuật:} Xây dựng kiến trúc module hóa, hỗ trợ theo dõi theo thời gian thực và triển khai hệ thống bằng Docker.
\item
  \textbf{Tính khả dụng:} Thiết kế giao diện phân quyền theo vai trò, giúp đơn giản hóa các quy trình nghiệp vụ chính cho tất cả các nhóm người dùng.
\end{itemize}
